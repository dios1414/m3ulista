<script>
const M3U_URL = "https://bit.ly/4ydeAZ8";

let canales = [];
let hlsInstance = null;

async function cargarLista() {
    const estado = document.getElementById("estado");
    try {
        const respuesta = await fetch(M3U_URL);
        if (!respuesta.ok) throw new Error("No se pudo descargar la lista");

        const texto = await respuesta.text();
        canales = analizarM3U(texto);

        poblarCategorias(canales);
        mostrarCanales(canales);

        estado.textContent = "Canales disponibles: " + canales.length;
    } catch (error) {
        console.error(error);
        estado.textContent = "No se pudo cargar la lista.";
        const mensaje = document.getElementById("error");
        mensaje.style.display = "block";
        mensaje.innerHTML =
            "⚠️ No se pudo acceder a la lista M3U.<br><br>" +
            "Si la URL funciona en un reproductor pero no aquí, " +
            "probablemente el servidor no permite solicitudes CORS.";
    }
}

function analizarM3U(texto) {
    const lineas = texto.split(/\r?\n/);
    const resultado = [];
    let informacion = null;

    for (let i = 0; i < lineas.length; i++) {
        const linea = lineas[i].trim();

        if (linea.startsWith("#EXTINF")) {
            informacion = {
                nombre: obtenerNombre(linea),
                logo: obtenerAtributo(linea, "tvg-logo"),
                grupo: obtenerAtributo(linea, "group-title") || "Sin Categoría"
            };
        } else if (linea && !linea.startsWith("#") && informacion) {
            resultado.push({
                nombre: informacion.nombre,
                logo: informacion.logo,
                grupo: informacion.grupo,
                url: linea
            });
            informacion = null;
        }
    }
    return resultado;
}

function obtenerNombre(linea) {
    const posicion = linea.lastIndexOf(",");
    return posicion === -1 ? "Canal" : linea.substring(posicion + 1).trim();
}

function obtenerAtributo(linea, atributo) {
    const expresion = new RegExp(atributo + '="([^"]*)"', "i");
    const resultado = linea.match(expresion);
    return resultado ? resultado[1] : "";
}

function poblarCategorias(lista) {
    const selector = document.getElementById("categoria");
    const grupos = [...new Set(lista.map(c => c.grupo))].sort();

    grupos.forEach(grupo => {
        const op = document.createElement("option");
        op.value = grupo;
        op.textContent = grupo;
        selector.appendChild(op);
    });
}

function mostrarCanales(lista) {
    const contenedor = document.getElementById("canales");
    contenedor.innerHTML = "";

    lista.forEach(function(canal) {
        const tarjeta = document.createElement("div");
        tarjeta.className = "card";
        tarjeta.tabIndex = 0;

        if (canal.logo) {
            const imagen = document.createElement("img");
            imagen.src = canal.logo;
            imagen.alt = "";
            imagen.onerror = function() { this.style.display = "none"; };
            tarjeta.appendChild(imagen);
        }

        const nombre = document.createElement("div");
        nombre.className = "nombre";
        nombre.textContent = canal.nombre;
        tarjeta.appendChild(nombre);

        tarjeta.onclick = function() {
            reproducir(canal.url, canal.nombre);
        };

        // Activar con la tecla Enter / OK del control remoto
        tarjeta.onkeydown = function(e) {
            if (e.key === "Enter") {
                reproducir(canal.url, canal.nombre);
            }
        };

        contenedor.appendChild(tarjeta);
    });
}

function reproducir(url, nombre) {
    const player = document.getElementById("player");
    const video = document.getElementById("video");
    const titulo = document.getElementById("tituloPlayer");

    titulo.textContent = "📺 " + nombre;
    player.style.display = "block";

    // Limpiar instancia anterior de HLS si existe
    if (hlsInstance) {
        hlsInstance.destroy();
        hlsInstance = null;
    }

    // Soporte reproducción HLS (.m3u8)
    if (Hls.isSupported() && url.includes(".m3u8")) {
        hlsInstance = new Hls();
        hlsInstance.loadSource(url);
        hlsInstance.attachMedia(video);
        hlsInstance.on(Hls.Events.MANIFEST_PARSED, function() {
            video.play();
        });
    } else if (video.canPlayType("application/vnd.apple.mpegurl")) {
        // Soporte nativo (Safari / iOS)
        video.src = url;
        video.play();
    } else {
        video.src = url;
        video.play().catch(err => console.log("Error al reproducir fuente directa", err));
    }

    player.scrollIntoView({ behavior: "smooth" });
    setTimeout(() => document.getElementById("volver").focus(), 300);
}

function cerrarPlayer() {
    const video = document.getElementById("video");
    
    video.pause();
    if (hlsInstance) {
        hlsInstance.destroy();
        hlsInstance = null;
    }
    video.removeAttribute("src");
    video.load();

    document.getElementById("player").style.display = "none";

    const primera = document.querySelector(".card");
    if (primera) primera.focus();
}

document.getElementById("volver").onclick = cerrarPlayer;

// Filtrado combinado (Buscador + Categoría)
function filtrar() {
    const texto = document.getElementById("busqueda").value.toLowerCase().trim();
    const categoria = document.getElementById("categoria").value;

    const filtrados = canales.filter(canal => {
        const coincideNombre = canal.nombre.toLowerCase().includes(texto);
        const coincideGrupo = categoria === "" || canal.grupo === categoria;
        return coincideNombre && coincideGrupo;
    });

    mostrarCanales(filtrados);
}

document.getElementById("busqueda").addEventListener("input", filtrar);
document.getElementById("categoria").addEventListener("change", filtrar);

// Botón ATRÁS / ESC / Teclas Smart TV
document.addEventListener("keydown", function(event) {
    if (event.key === "Escape" || event.key === "BrowserBack" || event.keyCode === 10009 || event.keyCode === 461) {
        const player = document.getElementById("player");
        if (player.style.display === "block") {
            cerrarPlayer();
            event.preventDefault();
        }
    }
});

window.onload = cargarLista;
